cmake_minimum_required(VERSION 3.16)
project(sqlon3_cpp LANGUAGES CXX)
add_library(sqlon3 Sqlon3.cpp Sqlon3Optimizer.cpp)
target_include_directories(sqlon3 PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(sqlon3 PUBLIC cxx_std_17)
add_executable(test_sqlon3 test_sqlon3.cpp)
target_link_libraries(test_sqlon3 PRIVATE sqlon3)
enable_testing()
add_test(NAME sqlon3_vectors COMMAND test_sqlon3)
set_tests_properties(sqlon3_vectors PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/..)

add_executable(test_optimization test_optimization.cpp)
target_link_libraries(test_optimization PRIVATE sqlon3)
add_test(NAME sqlon3_optimization COMMAND test_optimization)
set_tests_properties(sqlon3_optimization PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/..")
